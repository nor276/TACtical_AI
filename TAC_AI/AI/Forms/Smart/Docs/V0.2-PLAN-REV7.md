### Item 24 — Perimeter coverage cost (REV 4 — RESTORED via relocation, ships materially)

**REV 4 reframe:** REV 2's "drop and defer the relocate" was a sleight of hand — 6/10 of its own designers recommended the relocation to `StrategicValueFunction`. REV 4 ships the relocate IN P5, not "later."

**Action:**

1. Add to `StrategicValueFunction.cs`:
   ```csharp
   // Add to weight constants block at line ~18
   public static float WPerimeterCoverage = 0f;  // default OFF — flip to ~1.0 for defensive-perimeter rewarding plans

   // Add to Evaluate() body at line ~28
   + WPerimeterCoverage * PerimeterCoverage(state)

   // New term method
   /// <summary>
   /// Reward plans that maintain friendly-tech radial coverage around the team centroid.
   /// Returns mean cos-similarity between angular spread and uniform distribution
   /// over 2*pi: 1.0 means perfectly distributed (full perimeter), 0.0 means clustered.
   /// </summary>
   public static float PerimeterCoverage(StrategicState s)
   {
       if (s.Friendly.Count < 2) return 0f;
       Vector3 centroid = Vector3.zero;
       for (int i = 0; i < s.Friendly.Count; i++) centroid += s.Friendly[i].PositionMean;
       centroid /= s.Friendly.Count;
       // Compute angular dispersion in XZ plane (ground game).
       float sumCos = 0f, sumSin = 0f;
       for (int i = 0; i < s.Friendly.Count; i++)
       {
           Vector3 d = s.Friendly[i].PositionMean - centroid;
           float ang = Mathf.Atan2(d.z, d.x);
           sumCos += Mathf.Cos(ang);
           sumSin += Mathf.Sin(ang);
       }
       // Resultant vector length R: 0 = perfectly dispersed, 1 = perfectly clustered.
       // Coverage score = 1 - R.
       float R = Mathf.Sqrt(sumCos * sumCos + sumSin * sumSin) / s.Friendly.Count;
       return 1f - R;

   using System.Collections.Concurrent;
   namespace TAC_AI.AI.Forms.Smart.World
   {
       /// <summary>
       /// Per-tech tank-wide HpFraction sidecar via block-survival ratio.
       /// REV 7: matches the production multi-block HP semantic from TankAIHelper.cs:3267
       /// (DamageThreshold = (1 - blockC/maxBlockCount)*100). Tracks (current, max) per
       /// TechId; max ratchets upward to handle block additions. Read by Item 26
       /// RepairSupportGoalSource. Populated by main-thread ObserveWorldTechsIfDue.
       /// </summary>
       public sealed class HealthSidecar
       {
           private struct HpState { public int Current; public int Max; }
           private readonly ConcurrentDictionary<TechId, HpState> _state =
               new ConcurrentDictionary<TechId, HpState>();

           /// <summary>
           /// Record current block count for a tech. First call snapshots max;
           /// subsequent calls ratchet max upward if blockCount grows (handles in-game
           /// block additions). NEVER reduces max — handles per-design HP semantic
           /// (damage = lost blocks against the tech's high-water mark).
           /// </summary>
           public void Record(TechId id, int blockCount)
           {
               if (blockCount <= 0) return;
               _state.AddOrUpdate(id,
                   addValueFactory: _ => new HpState { Current = blockCount, Max = blockCount },
                   updateValueFactory: (_, s) => new HpState
                   {
                       Current = blockCount,
                       Max = blockCount > s.Max ? blockCount : s.Max
                   });
           }

           /// <summary>Tank-wide HpFraction in [0, 1]. Returns 1f for unknown techs (safe default).</summary>
           public float Get(TechId id)
           {
               HpState s;
               if (!_state.TryGetValue(id, out s) || s.Max <= 0) return 1f;
               float frac = (float)s.Current / s.Max;
               return frac < 0f ? 0f : (frac > 1f ? 1f : frac);
           }

           public void Forget(TechId id) { HpState _; _state.TryRemove(id, out _); }
           public void Clear() => _state.Clear();
       }
   }
   ```

public sealed class PatrolGoalSource : ISmartGoalSource
{
    public SmartIdentity Identity => SmartIdentity.Patrol;
    public const float ProvocationRadius = 80f;
    public const float MeanderRadius = 60f;
    public const float MeanderTimeScale = 0.02f;

    public TacticalGoal Produce(BeliefState ownBelief, VehicleModelSnapshot vehicle,
                                BeliefSnapshot beliefs, IdentityContext ctx)
    {
        // Inline hostile scan over BeliefSnapshot.ByTech.
        // REV 6 fix: use verified field names (b.Team, not b.Faction) per BeliefState.cs:98
        // and TacticalOptimizer.cs:141 pattern (pair.Value.Team.Equals(ownTeam)).
        BeliefState nearest = null;
        float bestSqr = ProvocationRadius * ProvocationRadius;
        foreach (var kv in beliefs.ByTech)
        {
            var b = kv.Value;
            if (b.Team.Equals(ownBelief.Team)) continue;       // skip friendlies (REV 6: b.Team, not b.Faction)
            if (b.Id.Equals(ctx.SelfTechId)) continue;          // skip self (REV 6: b.Id, not b.TechId)
            float d = (b.PositionMean - ownBelief.PositionMean).sqrMagnitude;
            if (d < bestSqr) { bestSqr = d; nearest = b; }
        }
        if (nearest != null)
        {
            // Pursue: drive-to with engagement velocity vector.
            // REV 6 fix: real TacticalGoal ctor per CostFunction.cs:20 — (pos, heading, vel, lookAheadSec).
            // No TacticalGoalKind enum exists in the codebase; the engagement intent is encoded
            // by setting velocity = heading * speed (mirrors HunterGoalSource pattern).
            Vector3 toHostile = nearest.PositionMean - ownBelief.PositionMean;
            Vector3 headingVec = toHostile.sqrMagnitude > 1e-4f ? toHostile.normalized : Vector3.forward;
            float heading = Mathf.Atan2(headingVec.x, headingVec.z);
            const float PursuitSpeed = 1f;
            const float LookAheadSec = 1.0f;
            Vector3 velocity = headingVec * PursuitSpeed * (1f / LookAheadSec) * 10f; // pursuit pace
            return new TacticalGoal(nearest.PositionMean, heading, velocity, LookAheadSec);
        }
        // Idle meander: deterministic Lissajous around SpawnAnchor seeded by TechId.
        // REV 6 fix: MonoClock.SecondsSinceStart() does not exist (real API per MonoClock.cs:22-50:
        // Now(), Seconds(from, to), SecondsDouble(from, to), TickFreq).
        // REV 7 doc correction (P-REV6-N5, 9/10): MonoClock.Now() wraps Stopwatch.GetTimestamp(),
        // which on Windows is QueryPerformanceCounter (SYSTEM-BOOT relative, not process-start).
        // For Lissajous we only use deltas across ticks (seed is per-tech-stable; t advances at a
        // fixed rate per second). The absolute origin is irrelevant — phase offset is determined
        // by `seed`, not by t's absolute value. Functionally correct as-is.
        float seed = (ctx.SelfTechId.Value & 0xFFFF) * 0.001f;
        float t = (float)(MonoClock.Now() * MonoClock.TickFreq) * MeanderTimeScale + seed;
        var offset = new Vector3(Mathf.Sin(t * 1.7f), 0f, Mathf.Cos(t * 1.1f)) * MeanderRadius;
        Vector3 meanderTarget = ctx.Stamp.SpawnAnchor + offset;
        Vector3 meanderHeadingVec = offset.sqrMagnitude > 1e-4f ? offset.normalized : Vector3.forward;
        float meanderHeading = Mathf.Atan2(meanderHeadingVec.x, meanderHeadingVec.z);
        const float MeanderSpeed = 0.5f;
        const float MeanderLookAheadSec = 1.0f;
        Vector3 meanderVelocity = (meanderTarget - ownBelief.PositionMean) / MeanderLookAheadSec * MeanderSpeed;
        return new TacticalGoal(meanderTarget, meanderHeading, meanderVelocity, MeanderLookAheadSec);
    }
