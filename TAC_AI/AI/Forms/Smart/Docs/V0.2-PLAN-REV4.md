   /// <summary>
   /// Reward plans that maintain friendly-tech radial coverage around the team centroid.
   /// Returns mean cos-similarity between angular spread and uniform distribution
   /// over 2*pi: 1.0 means perfectly distributed (full perimeter), 0.0 means clustered.
   /// </summary>
   public static float PerimeterCoverage(StrategicState s)
   {
       if (s.Friendly.Count < 2) return 0f;
       Vector3 centroid = Vector3.zero;
       for (int i = 0; i < s.Friendly.Count; i++) centroid += s.Friendly[i].PositionMean;
       centroid /= s.Friendly.Count;
       // Compute angular dispersion in XZ plane (ground game).
       float sumCos = 0f, sumSin = 0f;
       for (int i = 0; i < s.Friendly.Count; i++)
       {
           Vector3 d = s.Friendly[i].PositionMean - centroid;
           float ang = Mathf.Atan2(d.z, d.x);
           sumCos += Mathf.Cos(ang);
           sumSin += Mathf.Sin(ang);
       }
       // Resultant vector length R: 0 = perfectly dispersed, 1 = perfectly clustered.
       // Coverage score = 1 - R.
       float R = Mathf.Sqrt(sumCos * sumCos + sumSin * sumSin) / s.Friendly.Count;
       return 1f - R;
   }

            return new TacticalGoal(TacticalGoalKind.Engage,
                target: nearest.Value.PositionMean,
                heading: (nearest.Value.PositionMean - ownBelief.PositionMean).normalized,
                speed: 1f);
        }
        // Idle meander: deterministic Lissajous around SpawnAnchor seeded by TechId.
        // Pattern lifted from GathererGoalSource.cs:80-95 — substitute MeanderRadius/TimeScale.
        float seed = (ctx.SelfTechId.Value & 0xFFFF) * 0.001f;
        float t = (float)MonoClock.SecondsSinceStart() * MeanderTimeScale + seed;
        var offset = new Vector3(Mathf.Sin(t * 1.7f), 0f, Mathf.Cos(t * 1.1f)) * MeanderRadius;
        return new TacticalGoal(TacticalGoalKind.Move,
            target: ctx.Stamp.SpawnAnchor + offset,
            heading: offset.normalized,
            speed: 0.5f);
